<script>
const ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ".split("");

const SIGN_DESCRIPTIONS = {
A:"Make a fist with thumb to the side.",
B:"Four fingers up, thumb across palm.",
C:"Curve hand into a C shape.",
D:"Index up, other fingers touch thumb.",
E:"Curl fingers to touch thumb.",
F:"Index and thumb circle, others up.",
G:"Point sideways with thumb out.",
H:"Index and middle sideways.",
I:"Raise pinky only.",
J:"Draw J with pinky.",
K:"V shape with thumb between.",
L:"Form L with thumb and index.",
M:"Three fingers over thumb.",
N:"Two fingers over thumb.",
O:"All fingers form O.",
P:"Index downward with middle extended.",
Q:"Index and thumb downward.",
R:"Cross index and middle.",
S:"Fist with thumb over fingers.",
T:"Thumb between index and middle.",
U:"Index and middle together.",
V:"Index and middle V shape.",
W:"Three fingers spread.",
X:"Hook index finger.",
Y:"Thumb and pinky out.",
Z:"Draw Z in air."
};

const grid = document.getElementById("alphabetGrid");
const selectedLetterEl = document.getElementById("selectedLetter");
const descriptionEl = document.getElementById("description");

let selectedLetter = "A";
let avatarType = "male";
let quizMode = false;
let quizLetter = "";

function renderAlphabet(){
  ALPHABET.forEach(letter=>{
    const btn=document.createElement("div");
    btn.className="letter-card";
    btn.innerText=letter;
    btn.onclick=()=>selectLetter(letter,btn);
    grid.appendChild(btn);
  });
}

function selectLetter(letter,element){
  selectedLetter=letter;
  selectedLetterEl.innerText=letter;
  descriptionEl.innerText=SIGN_DESCRIPTIONS[letter];
  document.querySelectorAll(".letter-card").forEach(el=>el.classList.remove("active"));
  element.classList.add("active");
}

function replay(){
  alert("Replay animation for "+selectedLetter);
}

function toggleAvatar(){
  avatarType = avatarType==="male"?"female":"male";
  document.getElementById("avatarImg").style.filter=
    avatarType==="female"?"hue-rotate(40deg)":"none";
}

function toggleQuiz(){
  quizMode=!quizMode;
  document.getElementById("quizArea").style.display=
    quizMode?"block":"none";
  if(quizMode) nextQuiz();
}

function nextQuiz(){
  quizLetter = ALPHABET[Math.floor(Math.random()*26)];
  const options = new Set([quizLetter]);
  while(options.size<4){
    options.add(ALPHABET[Math.floor(Math.random()*26)]);
  }
  const optArray = Array.from(options);
  const container=document.getElementById("quizOptions");
  container.innerHTML="";
  optArray.forEach(opt=>{
    const btn=document.createElement("div");
    btn.className="quiz-btn";
    btn.innerText=opt;
    btn.onclick=()=>checkAnswer(opt,btn);
    container.appendChild(btn);
  });
  document.getElementById("quizResult").innerText="";
}

function checkAnswer(opt,btn){
  if(opt===quizLetter){
    btn.classList.add("correct");
    document.getElementById("quizResult").innerText="Correct!";
    setTimeout(nextQuiz,1500);
  }else{
    btn.classList.add("incorrect");
    document.getElementById("quizResult").innerText="Wrong! Correct: "+quizLetter;
  }
}

renderAlphabet();
selectLetter("A",document.querySelector(".letter-card"));
