    <script>
      const letters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ".split("");
      const grid = document.getElementById("grid");
      const desc = document.getElementById("desc");
      const selected = document.getElementById("selected");

      const descriptions = {
        A: "Make a fist with thumb to side.",
        B: "Four fingers up, thumb across palm.",
        C: "Curve hand into C shape.",
        D: "Index up, others touch thumb.",
        E: "Curl fingers touching thumb.",
        F: "Index and thumb circle.",
        G: "Point sideways.",
        H: "Two fingers sideways.",
        I: "Raise pinky.",
        J: "Draw J in air.",
        K: "V shape with thumb.",
        L: "Form L.",
        M: "Three fingers over thumb.",
        N: "Two fingers over thumb.",
        O: "Form O.",
        P: "Point downward.",
        Q: "Index and thumb down.",
        R: "Cross index and middle.",
        S: "Fist thumb over fingers.",
        T: "Thumb between fingers.",
        U: "Index and middle together.",
        V: "Victory shape.",
        W: "Three fingers up.",
        X: "Hook index.",
        Y: "Thumb and pinky out.",
        Z: "Draw Z."
      };

      let avatarType = "male";
      let quizLetter = "";

      letters.forEach(letter => {
        const div = document.createElement("div");
        div.className = "letter";
        div.innerText = letter;
        div.onclick = function () {
          document.querySelectorAll(".letter").forEach(el => el.classList.remove("active"));
          div.classList.add("active");
          selected.innerText = letter;
          desc.innerText = descriptions[letter];
        };
        grid.appendChild(div);
      });

      function replay() {
        alert("Replay animation");
      }

      function toggleAvatar() {
        avatarType = avatarType === "male" ? "female" : "male";
        document.getElementById("avatar").style.filter =
          avatarType === "female" ? "hue-rotate(40deg)" : "none";
      }

      function toggleQuiz() {
        const area = document.getElementById("quizArea");
        area.style.display = area.style.display === "none" ? "block" : "none";
      }
      var signQuestions = [
        { image: "A.png", correct: "A", options: ["A", "G", "F", "C"] },
        { image: "B.png", correct: "B", options: ["D", "B", "E", "H"] },
        { image: "C.png", correct: "C", options: ["C", "X", "Y", "T"] }
      ];

      var currentSign = 0;
      var signScore = 0;

      function loadSignQuestion() {

        var q = signQuestions[currentSign];

        document.getElementById("signAvatar").src =
          "/static/IMAGES/" + q.image;

        var optionsDiv = document.getElementById("signOptions");
        optionsDiv.innerHTML = "";

        for (var i = 0; i < q.options.length; i++) {

          var col = document.createElement("div");
          col.className = "col-6";

          var btn = document.createElement("div");
          btn.className = "card p-3 text-center";
          btn.style.cursor = "pointer";
          btn.innerText = q.options[i];

          btn.onclick = function () {
            checkSignAnswer(this.innerText);
          };

          col.appendChild(btn);
          optionsDiv.appendChild(col);
        }
      }

      function checkSignAnswer(selected) {

        var correct = signQuestions[currentSign].correct;

        if (selected === correct) {
          signScore++;
          document.getElementById("signResult").innerText = "Correct! ✅";
        } else {
          document.getElementById("signResult").innerText =
            "Wrong! Correct answer: " + correct;
        }

        setTimeout(function () {
          currentSign++;
          if (currentSign < signQuestions.length) {
            document.getElementById("signResult").innerText = "";
            loadSignQuestion();
          } else {
            document.querySelector(".sign-quiz").innerHTML =
              "<h2>Your Score: " + signScore + " / " + signQuestions.length + "</h2>";
          }
        }, 1000);
      }

      loadSignQuestion();
    </script>
